cmake_minimum_required(VERSION 3.20)
project(MillingAnalyzer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCASCADE 7.9.3 EXACT REQUIRED)
if(NOT TARGET TKOffset)
  message(FATAL_ERROR "OCCT install lacks TKOffset. Run cmake -P MillingAnalyzer/cmake/Dependencies.cmake to update the pinned dependency build.")
endif()
set(EIGEN_INCLUDE_DIR "" CACHE PATH "Eigen 3.4.0 source directory")
if(NOT EXISTS "${EIGEN_INCLUDE_DIR}/Eigen/Core")
  message(FATAL_ERROR "Set EIGEN_INCLUDE_DIR to the pinned Eigen 3.4.0 source directory")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/web/viewer.html")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/web/viewer.html" VIEWER_HEX HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," VIEWER_BYTES "${VIEWER_HEX}")
string(REPLACE "," ",\n" VIEWER_BYTES "${VIEWER_BYTES}")
configure_file(src/viewer_resource.h.in generated/viewer_resource.h @ONLY)
add_library(milling_geometry src/geometry.cpp src/stock.cpp src/program.cpp src/operation_plan.cpp)
target_include_directories(milling_geometry PUBLIC src "${EIGEN_INCLUDE_DIR}" "${OpenCASCADE_INCLUDE_DIR}")
target_link_libraries(milling_geometry PUBLIC TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKPrim TKMesh TKDESTEP TKBO TKOffset)
find_package(Threads REQUIRED)
get_filename_component(freza "${CMAKE_CURRENT_SOURCE_DIR}/../VisualStudio/FrezaEgm" ABSOLUTE)
add_library(milling_egm src/egm_session.cpp
 "${freza}/generated/egm3.pb.c" "${freza}/third_party/nanopb/pb_encode.c"
 "${freza}/third_party/nanopb/pb_decode.c" "${freza}/third_party/nanopb/pb_common.c")
target_include_directories(milling_egm PUBLIC src "${EIGEN_INCLUDE_DIR}" "${freza}/generated" "${freza}/third_party/nanopb")
target_link_libraries(milling_egm PUBLIC Threads::Threads)
if(WIN32)
 target_compile_definitions(milling_egm PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
 target_link_libraries(milling_egm PUBLIC ws2_32)
endif()
if(MSVC)
 target_compile_options(milling_egm PRIVATE /utf-8 /W4)
endif()
add_executable(MillingAnalyzer src/main.cpp src/server.cpp)
target_link_libraries(MillingAnalyzer PRIVATE milling_egm)
target_include_directories(MillingAnalyzer PRIVATE third_party)
target_link_libraries(MillingAnalyzer PRIVATE Threads::Threads)
target_include_directories(MillingAnalyzer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(MillingAnalyzer PRIVATE milling_geometry)
if(WIN32)
 target_link_libraries(MillingAnalyzer PRIVATE shell32 ws2_32)
endif()
if(WIN32)
 target_compile_definitions(milling_geometry PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MSVC)
 target_compile_options(milling_geometry PUBLIC /utf-8 /W4 /permissive-)
endif()
enable_testing()
add_executable(milling_egm_tests tests/egm_tests.cpp)
target_link_libraries(milling_egm_tests PRIVATE milling_egm)
add_test(NAME milling_egm_preflight COMMAND milling_egm_tests)
add_executable(milling_operation_tests tests/operation_tests.cpp)
target_link_libraries(milling_operation_tests PRIVATE milling_geometry)
add_test(NAME milling_operations COMMAND milling_operation_tests)
set_tests_properties(milling_operations PROPERTIES TIMEOUT 180)
add_executable(milling_program_tests tests/program_tests.cpp)
target_link_libraries(milling_program_tests PRIVATE milling_geometry)
add_test(NAME milling_program COMMAND milling_program_tests)
set_tests_properties(milling_program PROPERTIES TIMEOUT 180)
add_executable(milling_tests tests/tests.cpp)
target_link_libraries(milling_tests PRIVATE milling_geometry)
add_test(NAME milling_geometry COMMAND milling_tests)
set_tests_properties(milling_geometry PROPERTIES TIMEOUT 180)
add_test(NAME protected_abb COMMAND ${CMAKE_COMMAND} -P "${CMAKE_CURRENT_SOURCE_DIR}/../VisualStudio/FrezaEgm/tests/protected.cmake")
if(WIN32)
 foreach(target MillingAnalyzer milling_tests milling_program_tests milling_operation_tests)
  add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
   $<TARGET_RUNTIME_DLLS:${target}> $<TARGET_FILE_DIR:${target}> COMMAND_EXPAND_LISTS)
 endforeach()
endif()
add_custom_command(TARGET MillingAnalyzer POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
 "${CMAKE_CURRENT_SOURCE_DIR}/licenses" "$<TARGET_FILE_DIR:MillingAnalyzer>/licenses")
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
 add_test(NAME milling_local_api COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/http_test.py" $<TARGET_FILE:MillingAnalyzer> $<TARGET_FILE:milling_tests>)
 set_tests_properties(milling_local_api PROPERTIES TIMEOUT 180)
 add_test(NAME milling_egm_api_loopback COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/egm_http_test.py" $<TARGET_FILE:MillingAnalyzer> $<TARGET_FILE:milling_tests>)
 set_tests_properties(milling_egm_api_loopback PROPERTIES TIMEOUT 120)
endif()
