cmake_minimum_required(VERSION 3.20)
project(MillingAnalyzer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCASCADE 7.9.3 EXACT REQUIRED)
set(EIGEN_INCLUDE_DIR "" CACHE PATH "Eigen 3.4.0 source directory")
if(NOT EXISTS "${EIGEN_INCLUDE_DIR}/Eigen/Core")
  message(FATAL_ERROR "Set EIGEN_INCLUDE_DIR to the pinned Eigen 3.4.0 source directory")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/web/viewer.html")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/web/viewer.html" VIEWER_HEX HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," VIEWER_BYTES "${VIEWER_HEX}")
string(REPLACE "," ",\n" VIEWER_BYTES "${VIEWER_BYTES}")
configure_file(src/viewer_resource.h.in generated/viewer_resource.h @ONLY)
add_library(milling_geometry src/geometry.cpp)
target_include_directories(milling_geometry PUBLIC src "${EIGEN_INCLUDE_DIR}" "${OpenCASCADE_INCLUDE_DIR}")
target_link_libraries(milling_geometry PUBLIC TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKPrim TKMesh TKDESTEP)
find_package(Threads REQUIRED)
add_executable(MillingAnalyzer src/main.cpp src/server.cpp)
target_include_directories(MillingAnalyzer PRIVATE third_party)
target_link_libraries(MillingAnalyzer PRIVATE Threads::Threads)
target_include_directories(MillingAnalyzer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(MillingAnalyzer PRIVATE milling_geometry)
if(WIN32)
 target_link_libraries(MillingAnalyzer PRIVATE shell32 ws2_32)
endif()
if(WIN32)
 target_compile_definitions(milling_geometry PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MSVC)
 target_compile_options(milling_geometry PUBLIC /utf-8 /W4 /permissive-)
endif()
enable_testing()
add_executable(milling_tests tests/tests.cpp)
target_link_libraries(milling_tests PRIVATE milling_geometry)
add_test(NAME milling_geometry COMMAND milling_tests)
set_tests_properties(milling_geometry PROPERTIES TIMEOUT 180)
add_test(NAME protected_abb COMMAND ${CMAKE_COMMAND} -P "${CMAKE_CURRENT_SOURCE_DIR}/../VisualStudio/FrezaEgm/tests/protected.cmake")
if(WIN32)
 foreach(target MillingAnalyzer milling_tests)
  add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
   $<TARGET_RUNTIME_DLLS:${target}> $<TARGET_FILE_DIR:${target}> COMMAND_EXPAND_LISTS)
 endforeach()
endif()
add_custom_command(TARGET MillingAnalyzer POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
 "${CMAKE_CURRENT_SOURCE_DIR}/licenses" "$<TARGET_FILE_DIR:MillingAnalyzer>/licenses")
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
 add_test(NAME milling_local_api COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/http_test.py" $<TARGET_FILE:MillingAnalyzer> $<TARGET_FILE:milling_tests>)
 set_tests_properties(milling_local_api PROPERTIES TIMEOUT 180)
endif()
