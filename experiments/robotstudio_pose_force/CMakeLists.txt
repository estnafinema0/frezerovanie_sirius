cmake_minimum_required(VERSION 3.21)
project(MillingVirtualEgm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(EIGEN_INCLUDE_DIR "" CACHE PATH "Pinned Eigen 3.4.0")
get_filename_component(freza "${CMAKE_CURRENT_SOURCE_DIR}/../../VisualStudio/FrezaEgm" ABSOLUTE)
add_executable(MillingVirtualEgm virtual_egm.cpp
 "${freza}/generated/egm3.pb.c" "${freza}/third_party/nanopb/pb_encode.c"
 "${freza}/third_party/nanopb/pb_decode.c" "${freza}/third_party/nanopb/pb_common.c")
target_include_directories(MillingVirtualEgm PRIVATE "${EIGEN_INCLUDE_DIR}" "${freza}/generated" "${freza}/third_party/nanopb")
if(WIN32)
 target_compile_definitions(MillingVirtualEgm PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_link_libraries(MillingVirtualEgm PRIVATE ws2_32)
endif()
if(MSVC)
 target_compile_options(MillingVirtualEgm PRIVATE /utf-8)
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME virtual_pose_force_loopback COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_loopback.py" $<TARGET_FILE:MillingVirtualEgm>)
set_tests_properties(virtual_pose_force_loopback PROPERTIES TIMEOUT 60)
add_test(NAME protected_abb COMMAND ${CMAKE_COMMAND} -P "${freza}/tests/protected.cmake")
add_test(NAME single_task_pose_6510 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_pose6510.py" $<TARGET_FILE:MillingVirtualEgm>)
set_tests_properties(single_task_pose_6510 PROPERTIES TIMEOUT 20)
add_test(NAME pose_csv_protobuf COMMAND MillingVirtualEgm --check-poses "${CMAKE_CURRENT_SOURCE_DIR}/smoke.poses.csv")
